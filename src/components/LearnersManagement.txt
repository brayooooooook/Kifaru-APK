import React, { useState } from "react";
import { Learner } from "../types";
import { 
  UserPlus, Upload, Trash2, Edit2, Check, X, FileSpreadsheet, 
  Trash, RotateCcw, Search, AlertCircle, HelpCircle, FileText
} from "lucide-react";
import * as XLSX from "xlsx";
import { motion } from "motion/react";

interface LearnersManagementProps {
  learners: Learner[];
  token: string;
  onRefresh: () => void;
  onAlert: (msg: string, type: "success" | "error") => void;
}

export default function LearnersManagement({ 
  learners, 
  token, 
  onRefresh, 
  onAlert 
}: LearnersManagementProps) {
  const [searchTerm, setSearchTerm] = useState("");
  const [name, setName] = useState("");
  const [admissionNumber, setAdmissionNumber] = useState("");
  const [parentPhone, setParentPhone] = useState("");
  const [editingId, setEditingId] = useState<string | null>(null);
  const [editName, setEditName] = useState("");
  const [editAdmission, setEditAdmission] = useState("");
  const [editParentPhone, setEditParentPhone] = useState("");
  const [dragging, setDragging] = useState(false);
  const [importStatus, setImportStatus] = useState("");

  const handleAdd = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!name.trim()) return;

    try {
      const response = await fetch("/api/learners", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "Authorization": `Bearer ${token}`
        },
        body: JSON.stringify({ name, admissionNumber, parentPhone })
      });
      if (!response.ok) {
        throw new Error("Failed to add learner");
      }
      onAlert("Learner registered successfully", "success");
      setName("");
      setAdmissionNumber("");
      setParentPhone("");
      onRefresh();
    } catch (err: any) {
      onAlert(err.message, "error");
    }
  };

  const handleStartEdit = (learner: Learner) => {
    setEditingId(learner.id);
    setEditName(learner.name);
    setEditAdmission(learner.admissionNumber);
    setEditParentPhone(learner.parentPhone || "");
  };

  const handleCancelEdit = () => {
    setEditingId(null);
  };

  const handleSaveEdit = async (id: string) => {
    if (!editName.trim()) {
      onAlert("Name cannot be empty", "error");
      return;
    }
    try {
      const response = await fetch(`/api/learners/${id}`, {
        method: "PUT",
        headers: {
          "Content-Type": "application/json",
          "Authorization": `Bearer ${token}`
        },
        body: JSON.stringify({ name: editName, admissionNumber: editAdmission, parentPhone: editParentPhone })
      });
      if (!response.ok) {
        throw new Error("Failed to update learner");
      }
      onAlert("Learner updated successfully", "success");
      setEditingId(null);
      onRefresh();
    } catch (err: any) {
      onAlert(err.message, "error");
    }
  };

  const handleDelete = async (id: string) => {
    if (!confirm("Are you sure you want to delete this learner permanently? All their marks will be lost.")) {
      return;
    }
    try {
      const response = await fetch(`/api/learners/${id}`, {
        method: "DELETE",
        headers: {
          "Authorization": `Bearer ${token}`
        }
      });
      if (!response.ok) {
        throw new Error("Failed to delete learner");
      }
      onAlert("Learner deleted successfully", "success");
      onRefresh();
    } catch (err: any) {
      onAlert(err.message, "error");
    }
  };

  const handleClearAll = async () => {
    if (!confirm("CRITICAL WARNING: This will delete ALL learners and ALL marks. Are you absolutely sure?")) {
      return;
    }
    try {
      const response = await fetch("/api/learners/clear-all", {
        method: "POST",
        headers: {
          "Authorization": `Bearer ${token}`
        }
      });
      if (!response.ok) throw new Error("Failed to clear learners");
      onAlert("All records cleared", "success");
      onRefresh();
    } catch (err: any) {
      onAlert(err.message, "error");
    }
  };

  const handleResetDefaults = async () => {
    if (!confirm("This will reset the system to the default learners and re-fill sample marks. Existing changes will be replaced. Continue?")) {
      return;
    }
    try {
      const response = await fetch("/api/learners/reset-defaults", {
        method: "POST",
        headers: {
          "Authorization": `Bearer ${token}`
        }
      });
      if (!response.ok) throw new Error("Failed to reset learners");
      onAlert("Successfully restored default learners and sample marks", "success");
      onRefresh();
    } catch (err: any) {
      onAlert(err.message, "error");
    }
  };

  const processFile = (file: File) => {
    setImportStatus("Parsing file...");
    const reader = new FileReader();
    reader.onload = async (e) => {
      try {
        const bstr = e.target?.result;
        const workbook = XLSX.read(bstr, { type: "binary" });
        const sheetName = workbook.SheetNames[0];
        const worksheet = workbook.Sheets[sheetName];
        const data = XLSX.utils.sheet_to_json<any>(worksheet);

        if (data.length === 0) {
          throw new Error("Excel/CSV sheet is empty.");
        }

        // Parse list and map columns
        const parsedLearners = data.map((row: any) => {
          // Identify columns dynamically
          const nameCol = Object.keys(row).find(k => 
            k.toLowerCase().includes("name") || 
            k.toLowerCase().includes("student") || 
            k.toLowerCase().includes("learner")
          );
          const admCol = Object.keys(row).find(k => 
            k.toLowerCase().includes("adm") || 
            k.toLowerCase().includes("admission") || 
            k.toLowerCase().includes("no") || 
            k.toLowerCase().includes("number")
          );
          const phoneCol = Object.keys(row).find(k => 
            k.toLowerCase().includes("phone") || 
            k.toLowerCase().includes("contact") || 
            k.toLowerCase().includes("mobile") || 
            k.toLowerCase().includes("sms") || 
            k.toLowerCase().includes("parent")
          );

          const studentName = nameCol ? row[nameCol] : Object.values(row)[0];
          const studentAdm = admCol ? row[admCol] : (Object.values(row)[1] || "");
          const studentPhone = phoneCol ? row[phoneCol] : "";

          return {
            name: String(studentName || "").trim(),
            admissionNumber: String(studentAdm || "").trim(),
            parentPhone: String(studentPhone || "").trim()
          };
        }).filter(item => item.name.length > 0);

        if (parsedLearners.length === 0) {
          throw new Error("No valid learner names found. Ensure column header is 'Learner Name' or similar.");
        }

        setImportStatus(`Found ${parsedLearners.length} learners. Importing to server...`);

        const response = await fetch("/api/learners/bulk", {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            "Authorization": `Bearer ${token}`
          },
          body: JSON.stringify({ learners: parsedLearners })
        });

        if (!response.ok) {
          throw new Error("Server failed to import bulk learners");
        }

        const resData = await response.json();
        onAlert(`Successfully imported ${resData.count} learners!`, "success");
        setImportStatus("");
        onRefresh();
      } catch (err: any) {
        onAlert(err.message, "error");
        setImportStatus("");
      }
    };
    reader.onerror = () => {
      onAlert("Failed to read file", "error");
      setImportStatus("");
    };
    reader.readAsBinaryString(file);
  };

  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    if (e.target.files && e.target.files[0]) {
      processFile(e.target.files[0]);
    }
  };

  const handleDragOver = (e: React.DragEvent) => {
    e.preventDefault();
    setDragging(true);
  };

  const handleDragLeave = () => {
    setDragging(false);
  };

  const handleDrop = (e: React.DragEvent) => {
    e.preventDefault();
    setDragging(false);
    if (e.dataTransfer.files && e.dataTransfer.files[0]) {
      processFile(e.dataTransfer.files[0]);
    }
  };

  const filteredLearners = learners.filter(l => 
    l.name.toLowerCase().includes(searchTerm.toLowerCase()) ||
    l.admissionNumber.toLowerCase().includes(searchTerm.toLowerCase())
  );

  return (
    <div className="space-y-6">
      <div className="flex flex-col md:flex-row md:items-center justify-between gap-4">
        <div>
          <h2 className="text-2xl font-display font-bold text-gray-900">Learners Management</h2>
          <p className="text-sm text-gray-500">Register new learners, edit profiles, or import via Excel/CSV</p>
        </div>
        <div className="flex items-center gap-2.5">
          <button
            id="btn-restore-defaults"
            onClick={handleResetDefaults}
            className="flex items-center gap-1.5 py-1.5 px-3 bg-white border border-gray-200 text-gray-600 hover:text-[#1b365d] hover:border-[#1b365d] hover:bg-[#1b365d]/5 text-xs font-medium rounded-lg shadow-sm transition-all cursor-pointer"
          >
            <RotateCcw className="h-3.5 w-3.5" />
            Restore Defaults
          </button>
          <button
            id="btn-clear-all"
            onClick={handleClearAll}
            className="flex items-center gap-1.5 py-1.5 px-3 bg-white border border-red-100 text-red-500 hover:bg-rose-50 text-xs font-medium rounded-lg shadow-sm transition-all cursor-pointer"
          >
            <Trash className="h-3.5 w-3.5" />
            Clear All Records
          </button>
        </div>
      </div>

      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
        {/* Registration and Import Panel */}
        <div className="space-y-6 lg:col-span-1">
          {/* Quick Registration Card */}
          <div className="bg-white rounded-xl shadow-sm border border-gray-100 p-5 space-y-4">
            <h3 className="font-display font-semibold text-gray-900 flex items-center gap-2">
              <UserPlus className="h-5 w-5 text-blue-600" />
              Individual Registration
            </h3>
            <form onSubmit={handleAdd} className="space-y-3">
              <div>
                <label htmlFor="reg-name" className="block text-xs font-medium text-gray-500 mb-1">
                  Learner Full Name *
                </label>
                <input
                  id="reg-name"
                  type="text"
                  required
                  placeholder="e.g. John Austin Otieno"
                  value={name}
                  onChange={(e) => setName(e.target.value)}
                  className="w-full text-sm py-2 px-3 bg-gray-50 hover:bg-gray-50/50 focus:bg-white border border-gray-200 focus:border-blue-500 focus:ring-1 focus:ring-blue-500 rounded-lg transition-colors duration-150"
                />
              </div>
              <div>
                <label htmlFor="reg-adm" className="block text-xs font-medium text-gray-500 mb-1">
                  Admission Number (Optional)
                </label>
                <input
                  id="reg-adm"
                  type="text"
                  placeholder="e.g. MJS/G8B/001"
                  value={admissionNumber}
                  onChange={(e) => setAdmissionNumber(e.target.value)}
                  className="w-full text-sm py-2 px-3 bg-gray-50 hover:bg-gray-50/50 focus:bg-white border border-gray-200 focus:border-blue-500 focus:ring-1 focus:ring-blue-500 rounded-lg transition-colors duration-150"
                />
              </div>
              <div>
                <label htmlFor="reg-phone" className="block text-xs font-medium text-gray-500 mb-1">
                  Parent Phone Number (Optional)
                </label>
                <input
                  id="reg-phone"
                  type="text"
                  placeholder="e.g. 0712345678"
                  value={parentPhone}
                  onChange={(e) => setParentPhone(e.target.value)}
                  className="w-full text-sm py-2 px-3 bg-gray-50 hover:bg-gray-50/50 focus:bg-white border border-gray-200 focus:border-[#1b365d] focus:ring-1 focus:ring-[#1b365d] rounded-lg transition-colors duration-150"
                />
              </div>
              <button
                id="btn-register-learner"
                type="submit"
                className="w-full mt-2 py-2 px-4 bg-[#1b365d] hover:bg-[#152a4a] text-white font-semibold text-sm rounded-lg shadow-sm hover:shadow transition-all cursor-pointer flex items-center justify-center gap-1.5"
              >
                <Check className="h-4 w-4" />
                Register Learner
              </button>
            </form>
          </div>

          {/* Bulk Import Card */}
          <div className="bg-white rounded-xl shadow-sm border border-gray-100 p-5 space-y-4">
            <h3 className="font-display font-semibold text-gray-900 flex items-center gap-2">
              <Upload className="h-5 w-5 text-emerald-600" />
              Bulk Import (XLSX / CSV)
            </h3>
            
            <div
              id="drop-zone"
              onDragOver={handleDragOver}
              onDragLeave={handleDragLeave}
              onDrop={handleDrop}
              className={`border-2 border-dashed rounded-xl p-6 text-center transition-all ${
                dragging 
                  ? "border-emerald-500 bg-emerald-50/50" 
                  : "border-gray-200 hover:border-gray-300 bg-gray-50/30"
              }`}
            >
              <input
                id="file-uploader"
                type="file"
                accept=".xlsx,.xls,.csv"
                onChange={handleFileChange}
                className="hidden"
              />
              <label htmlFor="file-uploader" className="cursor-pointer space-y-2.5 block">
                <div className="mx-auto w-10 h-10 rounded-full bg-emerald-50 text-emerald-600 flex items-center justify-center">
                  <FileSpreadsheet className="h-5 w-5" />
                </div>
                <div>
                  <p className="text-sm font-semibold text-gray-700">Drag & drop sheet here</p>
                  <p className="text-xs text-gray-400 mt-0.5">or click to browse your computer</p>
                </div>
              </label>
            </div>

            {importStatus && (
              <div className="text-xs font-medium text-emerald-600 bg-emerald-50 border border-emerald-100 rounded-lg p-2.5 flex items-center gap-2">
                <span className="w-2 h-2 rounded-full bg-emerald-500 animate-ping shrink-0" />
                <span>{importStatus}</span>
              </div>
            )}

            <div className="bg-slate-50 border border-slate-100 rounded-xl p-3.5 space-y-2">
              <p className="text-xs font-semibold text-gray-600 flex items-center gap-1">
                <HelpCircle className="h-3.5 w-3.5 text-slate-400" />
                Spreadsheet Requirements:
              </p>
              <ul className="text-[11px] text-gray-500 space-y-1 list-disc pl-4 leading-relaxed">
                <li>Supports Microsoft Excel (<code className="font-mono bg-slate-100 px-0.5 rounded">.xlsx</code>, <code className="font-mono bg-slate-100 px-0.5 rounded">.xls</code>) or Comma Separated CSVs.</li>
                <li>First row should contain column headers (e.g. <strong className="text-gray-700">Learner Name</strong> and <strong className="text-gray-700">Admission Number</strong>).</li>
                <li>Automatically maps names to generate immediate teacher marks entry profiles.</li>
              </ul>
            </div>
          </div>
        </div>

        {/* Learner List Directory */}
        <div className="bg-white rounded-xl shadow-sm border border-gray-100 p-5 lg:col-span-2 space-y-4">
          <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
            <h3 className="font-display font-semibold text-gray-900">
              Grade 8 Blue Directory ({learners.length} Registered)
            </h3>
            <div className="relative rounded-lg shadow-sm max-w-xs w-full">
              <div className="absolute inset-y-0 left-0 pl-2.5 flex items-center pointer-events-none">
                <Search className="h-4 w-4 text-gray-400" />
              </div>
              <input
                id="search-input"
                type="text"
                placeholder="Search name or admission..."
                value={searchTerm}
                onChange={(e) => setSearchTerm(e.target.value)}
                className="block w-full pl-8 pr-3 py-1.5 text-xs bg-gray-50 hover:bg-gray-50/50 focus:bg-white border border-gray-200 focus:border-[#1b365d] rounded-lg text-gray-900 transition-all"
              />
            </div>
          </div>

          <div className="overflow-x-auto border border-gray-100 rounded-xl">
            <table className="min-w-full divide-y divide-gray-100">
              <thead className="bg-slate-50">
                <tr>
                  <th scope="col" className="px-4 py-3 text-left text-xs font-semibold text-gray-500 uppercase tracking-wider w-16">
                    No.
                  </th>
                  <th scope="col" className="px-4 py-3 text-left text-xs font-semibold text-gray-500 uppercase tracking-wider">
                    Learner Name
                  </th>
                  <th scope="col" className="px-4 py-3 text-left text-xs font-semibold text-gray-500 uppercase tracking-wider w-48">
                    Admission Number
                  </th>
                  <th scope="col" className="px-4 py-3 text-left text-xs font-semibold text-gray-500 uppercase tracking-wider w-48">
                    Parent Phone Number
                  </th>
                  <th scope="col" className="px-4 py-3 text-right text-xs font-semibold text-gray-500 uppercase tracking-wider w-24">
                    Actions
                  </th>
                </tr>
              </thead>
               <tbody className="bg-white dark:bg-slate-900 divide-y divide-gray-50 dark:divide-slate-800">
                 {filteredLearners.length === 0 ? (
                   <tr>
                     <td colSpan={5} className="px-4 py-8 text-center text-sm text-gray-400 dark:text-slate-500">
                       No matching learners found in Grade 8 Blue.
                     </td>
                   </tr>
                 ) : (
                   filteredLearners.map((learner, idx) => {
                     const isEven = idx % 2 === 0;
                     const rowBg = isEven ? "bg-white dark:bg-slate-900" : "bg-slate-50/40 dark:bg-slate-800/10";
                     return (
                       <tr key={learner.id} className={`${rowBg} hover:bg-slate-50/80 dark:hover:bg-slate-800/40 transition-colors`}>
                         <td className="px-4 py-3 whitespace-nowrap text-xs font-mono text-gray-400">
                           {idx + 1}
                         </td>
                         <td className="px-4 py-3 whitespace-nowrap">
                           {editingId === learner.id ? (
                             <input
                               id={`edit-name-${learner.id}`}
                               type="text"
                               value={editName}
                               onChange={(e) => setEditName(e.target.value)}
                               className="text-sm py-1 px-2.5 bg-white dark:bg-slate-800 border border-gray-300 dark:border-slate-700 focus:border-[#1b365d] focus:ring-1 focus:ring-[#1b365d] dark:text-slate-100 rounded-md w-full max-w-sm"
                             />
                           ) : (
                             <div className="text-sm font-medium text-gray-900 dark:text-slate-100">{learner.name}</div>
                           )}
                         </td>
                      <td className="px-4 py-3 whitespace-nowrap">
                        {editingId === learner.id ? (
                          <input
                            id={`edit-adm-${learner.id}`}
                            type="text"
                            value={editAdmission}
                            onChange={(e) => setEditAdmission(e.target.value)}
                            className="text-xs font-mono py-1 px-2.5 bg-white border border-gray-300 focus:border-[#1b365d] focus:ring-1 focus:ring-[#1b365d] rounded-md w-full max-w-xs"
                          />
                        ) : (
                          <span className="px-2 py-0.5 bg-gray-100 text-gray-600 rounded text-xs font-mono">
                            {learner.admissionNumber || "—"}
                          </span>
                        )}
                      </td>
                      <td className="px-4 py-3 whitespace-nowrap">
                        {editingId === learner.id ? (
                          <input
                            id={`edit-phone-${learner.id}`}
                            type="text"
                            value={editParentPhone}
                            onChange={(e) => setEditParentPhone(e.target.value)}
                            className="text-xs font-mono py-1 px-2.5 bg-white border border-gray-300 focus:border-[#1b365d] focus:ring-1 focus:ring-[#1b365d] rounded-md w-full max-w-xs"
                          />
                        ) : (
                          <span className="text-xs text-gray-700 font-mono">
                            {learner.parentPhone || "—"}
                          </span>
                        )}
                      </td>
                      <td className="px-4 py-3 whitespace-nowrap text-right text-sm font-medium">
                        {editingId === learner.id ? (
                          <div className="flex items-center justify-end gap-1.5">
                            <button
                              id={`btn-save-edit-${learner.id}`}
                              onClick={() => handleSaveEdit(learner.id)}
                              className="p-1 text-emerald-600 hover:bg-emerald-50 rounded transition-all cursor-pointer"
                              title="Save Changes"
                            >
                              <Check className="h-4 w-4" />
                            </button>
                            <button
                              id={`btn-cancel-edit-${learner.id}`}
                              onClick={handleCancelEdit}
                              className="p-1 text-gray-400 hover:bg-gray-100 rounded transition-all cursor-pointer"
                              title="Cancel"
                            >
                              <X className="h-4 w-4" />
                            </button>
                          </div>
                        ) : (
                          <div className="flex items-center justify-end gap-1.5">
                            <button
                              id={`btn-edit-${learner.id}`}
                              onClick={() => handleStartEdit(learner)}
                              className="p-1 text-[#1b365d] hover:bg-[#1b365d]/5 rounded transition-all cursor-pointer"
                              title="Edit Learner Info"
                            >
                              <Edit2 className="h-3.5 w-3.5" />
                            </button>
                            <button
                              id={`btn-delete-${learner.id}`}
                              onClick={() => handleDelete(learner.id)}
                              className="p-1 text-rose-500 hover:bg-rose-50 rounded transition-all cursor-pointer"
                              title="Delete Permanently"
                            >
                              <Trash2 className="h-3.5 w-3.5" />
                            </button>
                          </div>
                        )}
                      </td>
                    </tr>
                  );
                })
              )}
              </tbody>
            </table>
          </div>
        </div>
      </div>
    </div>
  );
}
